import TelegramBot from '@codebam/cf-workers-telegram-bot';

export default {
  async fetch(request: Request, env: Env) {
    const bot = new TelegramBot(env.TELEGRAM_TOKEN);

    await bot
      .command('start', async (ctx) =>
        ctx.reply('Running on Workers'))
      .onMessage(async (ctx) =>
        ctx.reply(`You said: ${ctx.text}`))
      .handle(request);

    return new Response('ok');
  },
};
